-- Roles table to define all user roles in the system
CREATE TABLE roles (
    role_id SERIAL PRIMARY KEY,
    role_name VARCHAR(50) UNIQUE NOT NULL,
    role_description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Users table with consistent INTEGER primary key
CREATE TABLE users (
    user_id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    phone VARCHAR(20),
    role_id INTEGER NOT NULL REFERENCES roles(role_id) ON DELETE RESTRICT,
    is_active BOOLEAN DEFAULT TRUE,
    last_login TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Regions table
CREATE TABLE regions (
    region_id SERIAL PRIMARY KEY,
    region_name VARCHAR(100) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Depots table
CREATE TABLE depots (
    depot_id SERIAL PRIMARY KEY,
    depot_name VARCHAR(100) NOT NULL,
    region_id INTEGER NOT NULL REFERENCES regions(region_id) ON DELETE CASCADE,
    address TEXT,
    contact_phone VARCHAR(20),
    latitude DECIMAL(10, 8),
    longitude DECIMAL(11, 8),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE dgm_technical (
    dgm_technical_id INTEGER PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
    appointment_date DATE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE dgm_operations (
    dgm_operations_id INTEGER PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
    appointment_date DATE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE regional_technical_officers (
    rto_id INTEGER PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
    region_id INTEGER NOT NULL REFERENCES regions(region_id) ON DELETE CASCADE,
    appointment_date DATE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE regional_operations_officers (
    roo_id INTEGER PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
    region_id INTEGER NOT NULL REFERENCES regions(region_id) ON DELETE CASCADE,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE depot_managers (

    depot_id INTEGER NOT NULL REFERENCES depots(depot_id) ON DELETE CASCADE,
    region_id INTEGER NOT NULL REFERENCES regions(region_id) ON DELETE CASCADE,
    appointment_date DATE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE depot_operation_managers (
    depot_op_manager_id INTEGER PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
    depot_id INTEGER NOT NULL REFERENCES depots(depot_id) ON DELETE CASCADE,
    region_id INTEGER NOT NULL REFERENCES regions(region_id) ON DELETE CASCADE,
    appointment_date DATE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE depot_engineers (
    depot_engineer_id INTEGER PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
    depot_id INTEGER NOT NULL REFERENCES depots(depot_id) ON DELETE CASCADE,
    region_id INTEGER NOT NULL REFERENCES regions(region_id) ON DELETE CASCADE,
    appointment_date DATE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE drivers (
    driver_id INTEGER PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
    depot_id INTEGER NOT NULL REFERENCES depots(depot_id) ON DELETE CASCADE,
    region_id INTEGER NOT NULL REFERENCES regions(region_id) ON DELETE CASCADE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE conductors (
    conductor_id INTEGER PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
    depot_id INTEGER NOT NULL REFERENCES depots(depot_id) ON DELETE CASCADE,
    region_id INTEGER NOT NULL REFERENCES regions(region_id) ON DELETE CASCADE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE passengers (
    passenger_id INTEGER PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE admins (
    admin_id INTEGER PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE emergency_contacts (

    name VARCHAR(100) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    relationship VARCHAR(50) DEFAULT 'N/A',
    email VARCHAR(100) DEFAULT 'N/A',
    is_primary BOOLEAN DEFAULT FALSE
);



CREATE TABLE passengers (
    passenger_id INTEGER PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
    date_of_birth DATE,
    gender VARCHAR(10) CHECK (gender IN ('male', 'female', 'other')),
    address TEXT,
    emergency_contact_name VARCHAR(100),
    emergency_contact_phone VARCHAR(20),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create trigger for updated_at
CREATE TRIGGER update_passengers_updated_at 
    BEFORE UPDATE ON passengers 
    FOR EACH ROW 
    EXECUTE FUNCTION update_updated_at_column();


CREATE TABLE buses (
    bus_id SERIAL PRIMARY KEY,
    registration_number VARCHAR(20) UNIQUE NOT NULL,
    depot_id INTEGER REFERENCES depots(depot_id) ON DELETE SET NULL,
    class VARCHAR(1) NOT NULL CHECK (class IN ('A', 'B', 'C', 'D')),
    manufacturer VARCHAR(100) NOT NULL,
    model VARCHAR(100),
    year INTEGER CHECK (year > 1900 AND year <= EXTRACT(YEAR FROM CURRENT_DATE)),
    mileage INTEGER DEFAULT 0 CHECK (mileage >= 0),
    status VARCHAR(20) NOT NULL DEFAULT 'Active' CHECK (status IN ('Active', 'In Service', 'Maintenance', 'Out of Service', 'Retired')),
    purchase_date DATE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_active BOOLEAN DEFAULT TRUE,
    is_deleted BOOLEAN DEFAULT FALSE
);

CREATE INDEX idx_buses_depot ON buses(depot_id);
CREATE INDEX idx_buses_status ON buses(status);
CREATE INDEX idx_buses_class ON buses(class);

-- Creating the bus_occupancy table
CREATE TABLE bus_occupancy (
    occupancy_id SERIAL PRIMARY KEY,
    bus_id INTEGER NOT NULL REFERENCES buses(bus_id) ON DELETE CASCADE,
    passenger_id INTEGER NOT NULL REFERENCES passengers(passenger_id) ON DELETE SET NULL,
    occupancy_level VARCHAR(20) NOT NULL CHECK (occupancy_level IN ('not_crowded', 'not_too_crowded', 'crowded', 'very_crowded')),
    latitude DOUBLE PRECISION NOT NULL CHECK (latitude BETWEEN -90 AND 90),
    longitude DOUBLE PRECISION NOT NULL CHECK (longitude BETWEEN -180 AND 180),
    confidence DOUBLE PRECISION CHECK (confidence BETWEEN 0 AND 100),
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE routes (
    route_id SERIAL PRIMARY KEY,
    route_number VARCHAR(10) UNIQUE NOT NULL,
    route_name VARCHAR(200) NOT NULL,
    start_location VARCHAR(100) NOT NULL,
    end_location VARCHAR(100) NOT NULL,
    distance_km DECIMAL(5,2),
    estimated_duration_minutes INTEGER,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create indexes
CREATE INDEX idx_routes_number ON routes(route_number);
CREATE INDEX idx_routes_active ON routes(is_active);
CREATE INDEX idx_routes_start_location ON routes(start_location);
CREATE INDEX idx_routes_end_location ON routes(end_location);

-- Create trigger to update updated_at timestamp
CREATE OR REPLACE FUNCTION update_routes_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_routes_updated_at
    BEFORE UPDATE ON routes
    FOR EACH ROW
    EXECUTE FUNCTION update_routes_updated_at();

CREATE TABLE IF NOT EXISTS public.route_stops
(
    stop_id integer NOT NULL GENERATED ALWAYS AS IDENTITY ( INCREMENT 1 START 1 MINVALUE 1 MAXVALUE 2147483647 CACHE 1 ),
    route_number character varying(10) COLLATE pg_catalog."default" NOT NULL,
    stop_name character varying(100) COLLATE pg_catalog."default" NOT NULL,
    stop_order integer NOT NULL,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT route_stops_pkey PRIMARY KEY (stop_id),
    CONSTRAINT route_stops_unique_route_stop UNIQUE (route_number, stop_order),
    CONSTRAINT route_stops_route_number_fkey FOREIGN KEY (route_number)
        REFERENCES public.routes (route_number) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE
);

CREATE INDEX idx_route_stops_route_number ON route_stops(route_number);
CREATE INDEX idx_route_stops_order ON route_stops(route_number, stop_order);

    checker_id INTEGER NOT NULL REFERENCES users(user_id) ON DELETE CASCADE, -- depot engineer who performed the check
    check_date DATE NOT NULL DEFAULT CURRENT_DATE,
    check_time TIME NOT NULL DEFAULT CURRENT_TIME,
    
    -- Main part checking flags
    engine BOOLEAN DEFAULT FALSE,
    brakes BOOLEAN DEFAULT FALSE,
    tires BOOLEAN DEFAULT FALSE,
    windows BOOLEAN DEFAULT FALSE,
    doors BOOLEAN DEFAULT FALSE,
    headlights BOOLEAN DEFAULT FALSE,
    signallights BOOLEAN DEFAULT FALSE,
    
    -- Overall checklist status
    is_checked BOOLEAN GENERATED ALWAYS AS (
        engine OR brakes OR tires OR windows OR doors OR headlights OR signallights
    ) STORED,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- Ensure one checklist per bus per day
    UNIQUE(bus_id, check_date)
);

-- Function to update updated_at timestamp
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Triggers for updated_at
CREATE TRIGGER update_communication_channels_updated_at
    BEFORE UPDATE ON communication_channels
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_messages_updated_at
    BEFORE UPDATE ON messages
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();

-- Function to get or create a direct channel between two users
CREATE OR REPLACE FUNCTION get_or_create_direct_channel(
    user1_id INTEGER,
    user2_id INTEGER
)
RETURNS INTEGER AS $$
DECLARE
    existing_channel_id INTEGER;
    new_channel_id INTEGER;
BEGIN
    -- Check if a direct channel already exists between these two users
    SELECT c.channel_id INTO existing_channel_id
    FROM communication_channels c
    WHERE c.channel_type = 'direct'
    AND EXISTS (
        SELECT 1 FROM channel_participants cp1
        WHERE cp1.channel_id = c.channel_id AND cp1.user_id = user1_id
    )
    AND EXISTS (
        SELECT 1 FROM channel_participants cp2
        WHERE cp2.channel_id = c.channel_id AND cp2.user_id = user2_id
    )
    AND (
        SELECT COUNT(*) FROM channel_participants cp
        WHERE cp.channel_id = c.channel_id
    ) = 2
    LIMIT 1;

    IF existing_channel_id IS NOT NULL THEN
        RETURN existing_channel_id;
    END IF;

    -- Create new channel
    INSERT INTO communication_channels (channel_type, created_by)
    VALUES ('direct', user1_id)
    RETURNING channel_id INTO new_channel_id;

    -- Add both participants
    INSERT INTO channel_participants (channel_id, user_id, can_send)
    VALUES 
        (new_channel_id, user1_id, TRUE),
        (new_channel_id, user2_id, TRUE);

    RETURN new_channel_id;
END;
$$ LANGUAGE plpgsql;
